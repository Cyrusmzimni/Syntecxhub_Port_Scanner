import socket
import sys
from datetime import datetime

def scan_ports(target_ip, port, timeout=1):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    result = sock.connect_ex((target_ip, port))
    sock.close()

    if result == 0:
        return "open"
    else:
        return "closed/filtered"
    
def main():
    target = input("Enter target IP or hostname (e.g. scanme.nmap.org): ").strip()
    port = int(input("Enter port number to scan (e.g. 80): ").strip())
 
    try:
        target_ip = socket.gethostbyname(target)
    except socket.gaierror:
        print(f"Could not resolve hostname: {target}")
        sys.exit(1)
 
    print(f"Scanning {target} ({target_ip}) on port {port}...")
    start_time = datetime.now()
 
    status = scan_ports(target_ip, port)
    print(f"Port {port}: {status}")
 
    end_time = datetime.now()
    print(f"Scan completed in {end_time - start_time}")
 
if __name__ == "__main__":
    main()
